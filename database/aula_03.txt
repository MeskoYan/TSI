--Aqui cria uma tabela curso
create table curso (
	codigo INT PRIMARY KEY,
	nome VARCHAR(30) NOT NULL,
	valor NUMERIC (6,2) CHECK (valor>=0 AND valor <= 1000) DEFAULT 100,
	email VARCHAR(30) UNIQUE,
	turno CHAR(1) CHECK (turno IN ('M','T','N','m','t','n'))
	--turno CHAR(1) CHECK (turno = 'M' OR turno = 'T' OR turno = 'N')
	--Pode fazer assim pra comentar
);

--Aqui cria a tabela aluno com a matricula auto-incrementando o nome obrigatorio e uma chave estrangeira
--para aluno para referenciar o curso por causa da relação 1:N em curso 1 -> n Aluno
CREATE TABLE aluno (
	matricula SERIAL PRIMARY KEY,
	nome varchar(50) NOT NULL,
	codigocurso INT,
	FOREIGN KEY (codigocurso) REFERENCES curso (codigo)
);

--Aqui cria uma tabela com chaves compostas (duas ou mais PKs)

	matricula SERIAL,
	nome varchar(50) NOT NULL,
	codigocurso INT,
	FOREIGN KEY (codigocurso) REFERENCES curso (codigo),
	PRIMARY KEY (matricula, nome)
